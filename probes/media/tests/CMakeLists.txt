cmake_minimum_required(VERSION 3.20)
project(ember_media_contracts C CXX)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec=63.1.102 libavformat=63.1.102 libavutil=61.1.102)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0=1.28.7 gstreamer-app-1.0=1.28.7)
add_executable(ffmpeg-contract ffmpeg.c)
target_link_libraries(ffmpeg-contract PRIVATE PkgConfig::AV)
add_executable(gstreamer-contract gstreamer.c)
target_link_libraries(gstreamer-contract PRIVATE PkgConfig::GST)
foreach(target ffmpeg-contract gstreamer-contract)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_test(NAME ffmpeg-round-trip COMMAND ffmpeg-contract "${CMAKE_CURRENT_BINARY_DIR}/frames.mkv")
set_tests_properties(ffmpeg-round-trip PROPERTIES FIXTURES_SETUP video TIMEOUT 35)
add_test(NAME ffmpeg-invalid-input COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/invalid-input.sh"
    "$<TARGET_FILE:ffmpeg-contract>" "${CMAKE_CURRENT_BINARY_DIR}/frames.mkv" "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(ffmpeg-invalid-input PROPERTIES FIXTURES_REQUIRED video TIMEOUT 35)
add_test(NAME gstreamer-app-contract COMMAND gstreamer-contract "${CMAKE_CURRENT_BINARY_DIR}/gstreamer.gray")
set_tests_properties(gstreamer-app-contract PROPERTIES TIMEOUT 50)
add_test(NAME media-build-guards COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/build-guards.sh"
    "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_tests_properties(media-build-guards PROPERTIES TIMEOUT 15)
option(VIDEOIO "Check installed OpenCV videoio against the same media profile" OFF)
if(VIDEOIO)
    find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc videoio)
    add_executable(videoio-contract videoio.cpp)
    target_link_libraries(videoio-contract PRIVATE ${OpenCV_LIBS})
    target_include_directories(videoio-contract PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_compile_features(videoio-contract PRIVATE cxx_std_17)
    target_compile_options(videoio-contract PRIVATE -Wall -Wextra -Werror)
    add_test(NAME opencv-videoio COMMAND videoio-contract
        "${CMAKE_CURRENT_BINARY_DIR}/frames.mkv" "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(opencv-videoio PROPERTIES FIXTURES_REQUIRED video TIMEOUT 65)
endif()
