cmake_minimum_required(VERSION 3.16)
project(ember_robotics_foundations_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 5.0.1 EXACT REQUIRED CONFIG)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs features geometry calib)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GPS REQUIRED IMPORTED_TARGET libgps=3.27.5)
enable_testing()
add_executable(eigen-contract eigen.cpp)
target_link_libraries(eigen-contract PRIVATE Eigen3::Eigen)
add_executable(opencv-contract opencv.cpp)
target_link_libraries(opencv-contract PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
target_include_directories(opencv-contract PRIVATE ${OpenCV_INCLUDE_DIRS})
add_executable(gpsd-contract gpsd.c)
target_compile_options(gpsd-contract PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gpsd-contract PRIVATE PkgConfig::GPS util m)
find_program(GPSD_EXECUTABLE gpsd REQUIRED)
add_test(NAME eigen-contract COMMAND eigen-contract)
add_test(NAME opencv-contract COMMAND opencv-contract)
add_test(NAME gpsd-contract COMMAND gpsd-contract "${GPSD_EXECUTABLE}")
add_test(NAME gpsd-missing-daemon COMMAND gpsd-contract /nonexistent/ember-gpsd)
set_tests_properties(gpsd-missing-daemon PROPERTIES WILL_FAIL TRUE TIMEOUT 8)
set_tests_properties(eigen-contract opencv-contract PROPERTIES TIMEOUT 30)
set_tests_properties(gpsd-contract PROPERTIES TIMEOUT 30)
add_test(NAME build-guards COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/check-build-guards.sh"
    "${CMAKE_CURRENT_SOURCE_DIR}/../build.sh")
set_tests_properties(build-guards PROPERTIES TIMEOUT 30)
add_test(NAME gpsd-missing-input COMMAND gpsd-contract "${GPSD_EXECUTABLE}" --no-feed)
set_tests_properties(gpsd-missing-input PROPERTIES WILL_FAIL TRUE TIMEOUT 30)
