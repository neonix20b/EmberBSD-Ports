cmake_minimum_required(VERSION 3.18)
project(ember_mcap_contract LANGUAGES CXX)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
  message(FATAL_ERROR "PROBE_PREFIX must be an absolute installed prefix")
endif()
find_path(MCAP_INCLUDE mcap/reader.hpp PATHS "${PROBE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_path(LZ4_INCLUDE lz4.h PATHS "${PROBE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_path(ZSTD_INCLUDE zstd.h PATHS "${PROBE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(LZ4_LIBRARY NAMES lz4 PATHS "${PROBE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd PATHS "${PROBE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
# A reused CMake cache must not redirect this consumer to a different installation.
foreach(path MCAP_INCLUDE LZ4_INCLUDE ZSTD_INCLUDE LZ4_LIBRARY ZSTD_LIBRARY)
  string(FIND "${${path}}" "${PROBE_PREFIX}/" position)
  if(NOT position EQUAL 0)
    message(FATAL_ERROR "${path} escaped the selected installation")
  endif()
endforeach()
add_executable(mcap-contract contract.cpp)
target_compile_features(mcap-contract PRIVATE cxx_std_17)
target_compile_options(mcap-contract PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(mcap-contract PRIVATE "${MCAP_INCLUDE}" "${LZ4_INCLUDE}" "${ZSTD_INCLUDE}")
target_link_libraries(mcap-contract PRIVATE "${LZ4_LIBRARY}" "${ZSTD_LIBRARY}")
foreach(mode none lz4 zstd)
  add_test(NAME mcap-${mode} COMMAND mcap-contract ${mode})
  set_tests_properties(mcap-${mode} PROPERTIES TIMEOUT 30)
endforeach()
