cmake_minimum_required(VERSION 3.16)
project(ember_fusion_contract LANGUAGES C)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
  message(FATAL_ERROR "PROBE_PREFIX must be an absolute installed prefix")
endif()
set(Fusion_DIR "${PROBE_PREFIX}/lib/cmake/Fusion")
find_package(Fusion 1.3.3 EXACT CONFIG REQUIRED PATHS "${Fusion_DIR}" NO_DEFAULT_PATH)
get_target_property(fusion_location Fusion::Fusion IMPORTED_LOCATION_RELEASE)
get_target_property(fusion_headers Fusion::Fusion INTERFACE_INCLUDE_DIRECTORIES)
foreach(path IN ITEMS "${fusion_location}" ${fusion_headers})
  string(FIND "${path}" "${PROBE_PREFIX}/" position)
  if(NOT position EQUAL 0)
    message(FATAL_ERROR "Fusion library or headers escaped the installed prefix: ${path}")
  endif()
endforeach()
add_executable(fusion-contract contract.c)
target_compile_features(fusion-contract PRIVATE c_std_99)
target_compile_options(fusion-contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(fusion-contract PRIVATE Fusion::Fusion)
foreach(case static rotation bias reset)
  add_test(NAME fusion-${case} COMMAND fusion-contract ${case})
  set_tests_properties(fusion-${case} PROPERTIES TIMEOUT 20)
endforeach()
