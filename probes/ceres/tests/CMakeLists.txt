cmake_minimum_required(VERSION 3.16)
project(ceres_installed_contract LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
  message(FATAL_ERROR "An absolute PROBE_PREFIX is required")
endif()
if(NOT DEFINED EIGEN_PREFIX)
  set(EIGEN_PREFIX "${PROBE_PREFIX}")
endif()
if(NOT IS_ABSOLUTE "${EIGEN_PREFIX}")
  message(FATAL_ERROR "An absolute EIGEN_PREFIX is required")
endif()
set(Eigen3_DIR "${EIGEN_PREFIX}/share/eigen3/cmake")
find_package(Eigen3 5.0.1 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${Eigen3_DIR}")
find_package(Ceres 2.2.0 EXACT CONFIG REQUIRED NO_DEFAULT_PATH
  PATHS "${PROBE_PREFIX}/lib/cmake/Ceres")
get_target_property(ceres_location Ceres::ceres IMPORTED_LOCATION_RELEASE)
string(FIND "${ceres_location}" "${PROBE_PREFIX}/lib/" in_prefix)
if(NOT in_prefix EQUAL 0)
  message(FATAL_ERROR "Ceres came from outside the installed probe: ${ceres_location}")
endif()
add_executable(ceres-contract contract.cpp)
target_link_libraries(ceres-contract PRIVATE Ceres::ceres)
foreach(case optimize invalid-options invalid-residual)
  add_test(NAME ceres-${case} COMMAND ceres-contract ${case})
  set_tests_properties(ceres-${case} PROPERTIES TIMEOUT 20)
endforeach()
