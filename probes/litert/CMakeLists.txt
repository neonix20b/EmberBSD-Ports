option(EMBER_BUILD_LM "Build LiteRT-LM CPU engine" ON)
foreach(dep abseil-cpp flatbuffers eigen cpuinfo ruy gemmlowp ml_dtypes farmhash fft2d xnnpack fp16_headers protobuf)
  string(TOUPPER "${dep}" upper)
  set("FETCHCONTENT_SOURCE_DIR_${upper}" "${EMBER_DEPS}/${dep}" CACHE PATH "" FORCE)
endforeach()
set(TENSORFLOW_SOURCE_DIR "${EMBER_DEPS}/tensorflow" CACHE PATH "" FORCE)

set(FP16_SOURCE_DIR "${EMBER_DEPS}/fp16_headers" CACHE PATH "" FORCE)
set(FXDIV_SOURCE_DIR "${EMBER_DEPS}/fxdiv" CACHE PATH "" FORCE)
set(CPUINFO_SOURCE_DIR "${EMBER_DEPS}/cpuinfo" CACHE PATH "" FORCE)
set(TFLITE_HOST_TOOLS_DIR "${EMBER_HOST_TOOLS}/flatbuffers" CACHE PATH "" FORCE)
set(EMBER_HOST_FLATC "${TFLITE_HOST_TOOLS_DIR}/flatc")
set(EMBER_HOST_PROTOC "${EMBER_HOST_TOOLS}/protobuf/protoc")
set(EMBER_PROTOBUF_SOURCE_DIR "${EMBER_DEPS}/protobuf")
set(LITERT_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(LITERT_ENABLE_NPU OFF CACHE BOOL "" FORCE)
set(LITERT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LITERT_BUILD_TENSOR OFF CACHE BOOL "" FORCE)

set(TFLITE_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)

set(FLATBUFFERS_BUILD_FLATC OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_INSTALL OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)

set(PTHREADPOOL_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(FXDIV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FXDIV_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
# Eigen is header-only. Use the common stable release without upstream's
# configure-time rewriting of its documentation and benchmark files.
add_library(eigen INTERFACE)
add_library(Eigen3::Eigen ALIAS eigen)
target_include_directories(eigen INTERFACE "${EMBER_DEPS}/eigen")
target_compile_definitions(eigen INTERFACE EIGEN_MPL2_ONLY)
set(eigen_POPULATED TRUE)
set(EIGEN_INCLUDE_DIRS "${EMBER_DEPS}/eigen")
set(LITERT_XNNPACK_INCLUDE_DIR "${EMBER_DEPS}/xnnpack/include")
set(LITERT_PTHREADPOOL_INCLUDE_DIR "${EMBER_DEPS}/pthreadpool/include")
add_subdirectory("${EMBER_LITERT_SOURCE_DIR}/litert" litert)
foreach(runtime_target litert_runtime litert_runtime_c_api_shared_lib)
  target_include_directories(${runtime_target} PUBLIC "$<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/include>")
endforeach()
if(EMBER_BUILD_LM)
  set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "" FORCE)
  set(protobuf_BUILD_LIBPROTOC OFF CACHE BOOL "" FORCE)
  set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
  set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
  set(protobuf_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  # The LiteRT profiling dependency has already created protobuf targets.
  set(RE2_INSTALL OFF CACHE BOOL "" FORCE)
  set(RE2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${EMBER_DEPS}/re2" re2)
  set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
  set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
  include("${CMAKE_CURRENT_SOURCE_DIR}/../litert-lm/cmake/sentencepiece.cmake")
  ember_add_sentencepiece("${EMBER_DEPS}/sentencepiece")
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  add_subdirectory("${EMBER_DEPS}/json" json)
  find_package(ZLIB REQUIRED)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../litert-lm" litert-lm)
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.cmake" OPTIONAL)
add_custom_target(ember_litert_bundle DEPENDS litert_runtime_c_api_shared_lib)
if(TARGET ember-litert-lm)
  add_dependencies(ember_litert_bundle ember-litert-lm ember-gated-delta-test)

if(TARGET ember-litert-runtime-tests)
  add_dependencies(ember_litert_bundle ember-litert-runtime-tests)

install(TARGETS litert_runtime_c_api_shared_lib
  LIBRARY DESTINATION lib COMPONENT EmberLiteRt)
install(DIRECTORY "${EMBER_LITERT_SOURCE_DIR}/litert/c"
                  "${EMBER_LITERT_SOURCE_DIR}/litert/cc"
  DESTINATION include/litert COMPONENT EmberLiteRt FILES_MATCHING PATTERN "*.h")
install(DIRECTORY "${EMBER_LITERT_SOURCE_DIR}/litert/vendors/"
  DESTINATION include/litert/vendors COMPONENT EmberLiteRt
  FILES_MATCHING REGEX "/litert_[^/]*\\.h$")

  DESTINATION include/litert/core COMPONENT EmberLiteRt)
install(FILES "${CMAKE_BINARY_DIR}/include/litert/build_common/build_config.h"
  DESTINATION include/litert/build_common COMPONENT EmberLiteRt)

  DESTINATION lib/cmake/EmberLiteRt COMPONENT EmberLiteRt)
install(FILES README.md PROVENANCE.md VALIDATION.md sources.tsv dependencies.tsv
  DESTINATION share/doc/ember-litert COMPONENT EmberLiteRt)
install(FILES LICENSE DESTINATION share/licenses/ember-litert COMPONENT EmberLiteRt)
install(FILES "${EMBER_LITERT_SOURCE_DIR}/LICENSE"
  DESTINATION share/licenses/ember-litert RENAME LiteRT-LICENSE COMPONENT EmberLiteRt)
# Keep licenses for every bundled source dependency alongside the runtime.
file(GLOB _dependency_dirs "${EMBER_DEPS}/*")
foreach(dependency IN LISTS _dependency_dirs)
  if(IS_DIRECTORY "${dependency}")
    get_filename_component(name "${dependency}" NAME)
    file(GLOB notices "${dependency}/LICENSE*" "${dependency}/COPYING*" "${dependency}/NOTICE*")
    foreach(notice IN LISTS notices)
      if(NOT IS_DIRECTORY "${notice}")
        install(FILES "${notice}" DESTINATION "share/licenses/ember-litert/${name}" COMPONENT EmberLiteRt)
      endif()
    endforeach()
  endif()
endforeach()
install(FILES "${EMBER_DEPS}/sentencepiece/third_party/darts_clone/LICENSE"
  DESTINATION share/licenses/ember-litert/sentencepiece/third_party/darts_clone COMPONENT EmberLiteRt)

  DESTINATION share/licenses/ember-litert/protobuf/third_party/utf8_range COMPONENT EmberLiteRt)

    DESTINATION share/doc/ember-litert-lm COMPONENT EmberLiteRt)
endif()
