cmake_minimum_required(VERSION 3.22)
project(ember_volk_contract LANGUAGES C CXX)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
    message(FATAL_ERROR "PROBE_PREFIX must be an absolute installed prefix")
endif()
find_package(Volk 3.3.0 EXACT CONFIG REQUIRED
    PATHS "${PROBE_PREFIX}/lib/cmake/volk" NO_DEFAULT_PATH)
get_target_property(volk_location Volk::volk IMPORTED_LOCATION_RELEASE)
string(FIND "${volk_location}" "${PROBE_PREFIX}/" position)
if(NOT position EQUAL 0)
    message(FATAL_ERROR "VOLK escaped the selected installation")
endif()
add_executable(volk-contract contract.cc)
target_compile_features(volk-contract PRIVATE cxx_std_17)
target_compile_options(volk-contract PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(volk-contract PRIVATE Volk::volk)
foreach(kernel dot multiply magnitude)
    add_test(NAME volk-${kernel}-dispatch COMMAND volk-contract ${kernel})
    add_test(NAME volk-${kernel}-generic COMMAND volk-contract ${kernel})
    set_tests_properties(volk-${kernel}-dispatch volk-${kernel}-generic PROPERTIES TIMEOUT 30)
    set_tests_properties(volk-${kernel}-generic PROPERTIES ENVIRONMENT "VOLK_GENERIC=1")
endforeach()
add_executable(volk-math-c math-header.c)
target_compile_features(volk-math-c PRIVATE c_std_17)
target_include_directories(volk-math-c PRIVATE "${PROBE_PREFIX}/include")
target_link_libraries(volk-math-c PRIVATE m)
configure_file(math-header.c "${CMAKE_CURRENT_BINARY_DIR}/math-header.cc" COPYONLY)
add_executable(volk-math-cxx "${CMAKE_CURRENT_BINARY_DIR}/math-header.cc")
target_compile_features(volk-math-cxx PRIVATE cxx_std_17)
target_include_directories(volk-math-cxx PRIVATE "${PROBE_PREFIX}/include")
foreach(language c cxx)
    add_test(NAME volk-math-header-${language} COMMAND volk-math-${language})
    set_tests_properties(volk-math-header-${language} PROPERTIES TIMEOUT 10)
endforeach()
add_test(NAME volk-profiler-smoke COMMAND "${CMAKE_COMMAND}"
    "-DPROFILER=${PROBE_PREFIX}/bin/volk_profile"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/profiler.cmake")
set_tests_properties(volk-profiler-smoke PROPERTIES TIMEOUT 30)
