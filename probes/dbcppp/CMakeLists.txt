cmake_minimum_required(VERSION 3.16)
project(emberbsd_dbcppp_probe LANGUAGES CXX)
include(TestBigEndian)
if(NOT EXISTS "${SOURCE}/src/libdbcppp/DBCX3.cpp" OR NOT EXISTS "${BOOST_HEADERS}/boost/version.hpp")
  message(FATAL_ERROR "Verified upstream SOURCE and BOOST_HEADERS are required")
endif()
file(STRINGS "${BOOST_HEADERS}/boost/version.hpp" boost_version REGEX "^#define BOOST_VERSION ")
if(NOT boost_version STREQUAL "#define BOOST_VERSION 109100")
  message(FATAL_ERROR "Boost 1.91.0 headers required")
endif()
test_big_endian(big)
if(big)
  set(BYTE_ORDER Big)
else()
  set(BYTE_ORDER Little)
endif()
configure_file("${SOURCE}/src/libdbcppp/EndianConfig.h.in" "${SOURCE}/src/libdbcppp/EndianConfig.h" @ONLY)
file(GLOB sources CONFIGURE_DEPENDS "${SOURCE}/src/libdbcppp/*.cpp")
list(FILTER sources EXCLUDE REGEX "/KCD2Network.cpp$")
add_library(dbcppp STATIC ${sources})
target_compile_features(dbcppp PUBLIC cxx_std_17)
target_compile_definitions(dbcppp PRIVATE DBCPPP_EXPORT)
target_include_directories(dbcppp PRIVATE "${SOURCE}/include" "${SOURCE}/src" "${BOOST_HEADERS}")
install(TARGETS dbcppp ARCHIVE DESTINATION lib)
install(DIRECTORY "${SOURCE}/include/dbcppp" DESTINATION include)
