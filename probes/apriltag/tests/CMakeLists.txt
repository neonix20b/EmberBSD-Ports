cmake_minimum_required(VERSION 3.16)
project(ember_apriltag_contract LANGUAGES C)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
  message(FATAL_ERROR "PROBE_PREFIX must be an absolute installed prefix")
endif()
# Bind the installed upstream package explicitly; never silently use a system copy.
set(apriltag_DIR "${PROBE_PREFIX}/lib/apriltag/cmake")
find_package(apriltag 3.4.5 EXACT CONFIG REQUIRED PATHS "${apriltag_DIR}" NO_DEFAULT_PATH)
get_target_property(apriltag_location apriltag::apriltag IMPORTED_LOCATION_RELEASE)
string(FIND "${apriltag_location}" "${PROBE_PREFIX}/" position)
if(NOT position EQUAL 0)
  message(FATAL_ERROR "AprilTag escaped the selected installation")
endif()
add_executable(apriltag-contract contract.c)
target_compile_features(apriltag-contract PRIVATE c_std_99)
target_compile_options(apriltag-contract PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(apriltag-contract PRIVATE apriltag::apriltag)
add_test(NAME apriltag-detection-pose COMMAND apriltag-contract)
set_tests_properties(apriltag-detection-pose PROPERTIES TIMEOUT 30)
add_executable(apriltag-ctype-contract ctype.c)
target_compile_features(apriltag-ctype-contract PRIVATE c_std_99)
target_compile_options(apriltag-ctype-contract PRIVATE -Wall -Wextra -Werror=char-subscripts)
# Treat installed headers as ordinary headers so the original ctype misuse is diagnosed.
set_target_properties(apriltag-ctype-contract PROPERTIES NO_SYSTEM_FROM_IMPORTED TRUE)
target_link_libraries(apriltag-ctype-contract PRIVATE apriltag::apriltag)
add_test(NAME apriltag-ctype-regression COMMAND apriltag-ctype-contract)
set_tests_properties(apriltag-ctype-regression PROPERTIES TIMEOUT 10)
