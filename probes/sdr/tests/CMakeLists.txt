cmake_minimum_required(VERSION 3.20)
project(ember_sdr_contracts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SoapySDR CONFIG REQUIRED)
if(NOT IS_ABSOLUTE "${SDR_PREFIX}" OR NOT EXISTS "${SDR_PREFIX}/lib/libiio.so.0")
  message(FATAL_ERROR "SDR_PREFIX must name the profile's installed prefix")
endif()
add_executable(make-iq make-iq.c)
add_executable(compat-contract compat-contract.c)
target_include_directories(compat-contract PRIVATE "${SDR_PREFIX}/include/iio-compat-0")
target_link_libraries(compat-contract PRIVATE "${SDR_PREFIX}/lib/libiio.so.0")
add_executable(soapy-contract soapy-contract.cpp)
target_link_libraries(soapy-contract PRIVATE SoapySDR)
if(NOT EXISTS "${LIBIIO_SOURCE}/iiod-emu/network.c")
  message(FATAL_ERROR "LIBIIO_SOURCE must name the patched pinned libiio source")
endif()
add_executable(loopback loopback.c "${LIBIIO_SOURCE}/iiod-emu/network.c")
target_include_directories(loopback PRIVATE "${LIBIIO_SOURCE}/iiod-emu" "${SDR_PREFIX}/include")
find_package(Threads REQUIRED)
find_library(IIO1_LIBRARY NAMES iio PATHS "${SDR_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(network-errors network-errors.cpp)
target_include_directories(network-errors PRIVATE "${SDR_PREFIX}/include")
target_link_libraries(network-errors PRIVATE "${IIO1_LIBRARY}" Threads::Threads)
add_executable(stream-error stream-error.cpp)
target_link_libraries(stream-error PRIVATE SoapySDR)
