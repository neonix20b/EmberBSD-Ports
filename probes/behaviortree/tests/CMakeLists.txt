cmake_minimum_required(VERSION 3.16)
project(behaviortree_installed_contract LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
  message(FATAL_ERROR "An absolute PROBE_PREFIX is required")
endif()
find_package(Threads REQUIRED)
find_package(behaviortree_cpp CONFIG REQUIRED NO_DEFAULT_PATH
  PATHS "${PROBE_PREFIX}/lib/cmake/behaviortree_cpp")
get_target_property(bt_location BT::behaviortree_cpp IMPORTED_LOCATION_RELEASE)
string(FIND "${bt_location}" "${PROBE_PREFIX}/lib/" in_prefix)
if(NOT in_prefix EQUAL 0)
  message(FATAL_ERROR "BehaviorTree came from outside the installed probe: ${bt_location}")
endif()
add_executable(behaviortree-contract contract.cpp)
target_link_libraries(behaviortree-contract PRIVATE BT::behaviortree_cpp Threads::Threads)
foreach(case success failure cancel timeout malformed logging)
  add_test(NAME behaviortree-${case} COMMAND behaviortree-contract ${case})
  set_tests_properties(behaviortree-${case} PROPERTIES TIMEOUT 15)
endforeach()
