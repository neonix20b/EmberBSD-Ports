cmake_minimum_required(VERSION 3.18)
project(ember_fftw_contract C)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
    message(FATAL_ERROR "An absolute PROBE_PREFIX is required")
endif()
unset(FFTW_INCLUDE CACHE)
find_path(FFTW_INCLUDE fftw3.h PATHS "${PROBE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
foreach(kind IN ITEMS double float)
    if(kind STREQUAL "float")
        set(library fftw3f)
    else()
        set(library fftw3)
    endif()
    unset(LIB_${kind} CACHE)
    unset(THREAD_${kind} CACHE)
    find_library(LIB_${kind} ${library} PATHS "${PROBE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
    find_library(THREAD_${kind} ${library}_threads PATHS "${PROBE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
    add_executable(fftw-${kind} contract.c)
    target_compile_features(fftw-${kind} PRIVATE c_std_99)
    target_compile_options(fftw-${kind} PRIVATE -Wall -Wextra -Werror)
    target_include_directories(fftw-${kind} PRIVATE "${FFTW_INCLUDE}")
    target_link_libraries(fftw-${kind} PRIVATE "${THREAD_${kind}}" "${LIB_${kind}}" m)
    if(kind STREQUAL "float")
        target_compile_definitions(fftw-${kind} PRIVATE USE_FLOAT)
    endif()
    add_test(NAME fftw-${kind} COMMAND fftw-${kind})
    set_tests_properties(fftw-${kind} PROPERTIES TIMEOUT 30)
endforeach()
