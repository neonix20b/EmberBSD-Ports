cmake_minimum_required(VERSION 3.16)
project(gtsam_installed_contract LANGUAGES CXX)
enable_testing()
add_executable(gtsam-matrix-check matrix-check.cpp)
add_test(NAME gtsam-matrix-check COMMAND gtsam-matrix-check)
set_tests_properties(gtsam-matrix-check PROPERTIES TIMEOUT 5)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(prefix PROBE_PREFIX EIGEN_PREFIX)
  if(NOT IS_ABSOLUTE "${${prefix}}")
    message(FATAL_ERROR "An absolute ${prefix} is required")
  endif()
endforeach()
set(Eigen3_DIR "${EIGEN_PREFIX}/share/eigen3/cmake")
find_package(Eigen3 5.0.1 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${Eigen3_DIR}")
target_link_libraries(gtsam-matrix-check PRIVATE Eigen3::Eigen)
target_compile_features(gtsam-matrix-check PRIVATE cxx_std_17)
find_package(GTSAM 4.3.0 EXACT CONFIG REQUIRED NO_DEFAULT_PATH
  PATHS "${PROBE_PREFIX}/lib/cmake/GTSAM")
get_target_property(location gtsam IMPORTED_LOCATION_RELEASE)
string(FIND "${location}" "${PROBE_PREFIX}/lib/" in_prefix)
if(NOT in_prefix EQUAL 0)
  message(FATAL_ERROR "GTSAM came from outside the installed probe: ${location}")
endif()
add_executable(gtsam-contract contract.cpp)
target_link_libraries(gtsam-contract PRIVATE gtsam)
foreach(case linear-fusion pose-graph underconstrained missing-state)
  add_test(NAME gtsam-${case} COMMAND gtsam-contract ${case})
  set_tests_properties(gtsam-${case} PROPERTIES TIMEOUT 30)
endforeach()
