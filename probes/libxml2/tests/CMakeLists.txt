cmake_minimum_required(VERSION 3.20)
project(ember_libxml2_contract LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${XML_PREFIX}")
  message(FATAL_ERROR "XML_PREFIX must be absolute")
endif()
find_package(libxml2 2.15.4 EXACT CONFIG REQUIRED PATHS "${XML_PREFIX}/lib/cmake/libxml2" NO_DEFAULT_PATH)
add_executable(xml-contract contract.c)
target_link_libraries(xml-contract PRIVATE LibXml2::LibXml2 ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME installed-xml-contract COMMAND xml-contract "${XML_PREFIX}")
set_tests_properties(installed-xml-contract PROPERTIES TIMEOUT 15)
