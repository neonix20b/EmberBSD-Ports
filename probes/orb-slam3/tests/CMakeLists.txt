cmake_minimum_required(VERSION 3.20)
project(ember_orb_contract LANGUAGES CXX)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}" OR NOT IS_ABSOLUTE "${COMMON_PREFIX}")
  message(FATAL_ERROR "PROBE_PREFIX and COMMON_PREFIX must be absolute installations")
endif()
set(OPENCV_PREFIX "${COMMON_PREFIX}" CACHE PATH "Selected common OpenCV installation")
if(NOT IS_ABSOLUTE "${OPENCV_PREFIX}")
  message(FATAL_ERROR "OPENCV_PREFIX must be an absolute installation")
endif()
set(Eigen3_DIR "${COMMON_PREFIX}/share/eigen3/cmake")
set(OpenCV_DIR "${OPENCV_PREFIX}/lib/cmake/opencv5")
set(ORB_SLAM3_DIR "${PROBE_PREFIX}/lib/cmake/ORB_SLAM3")
find_package(ORB_SLAM3 1.0.0 EXACT CONFIG REQUIRED PATHS "${ORB_SLAM3_DIR}" NO_DEFAULT_PATH)
get_target_property(eigen_headers Eigen3::Eigen INTERFACE_INCLUDE_DIRECTORIES)
foreach(path IN ITEMS ${eigen_headers})
  string(FIND "${path}" "${COMMON_PREFIX}/" position)
  if(NOT position EQUAL 0)
    message(FATAL_ERROR "Eigen headers escaped COMMON_PREFIX: ${path}")
  endif()
endforeach()
foreach(path IN ITEMS ${OpenCV_INCLUDE_DIRS})
  string(FIND "${path}" "${OPENCV_PREFIX}/" position)
  if(NOT position EQUAL 0)
    message(FATAL_ERROR "OpenCV headers escaped OPENCV_PREFIX: ${path}")
  endif()
endforeach()
foreach(target ORB_SLAM3 DBoW2 g2o)
  get_target_property(location ORB_SLAM3::${target} IMPORTED_LOCATION_RELEASE)
  get_target_property(headers ORB_SLAM3::${target} INTERFACE_INCLUDE_DIRECTORIES)
  foreach(path IN ITEMS "${location}" ${headers})
    string(FIND "${path}" "${PROBE_PREFIX}/" position)
    if(NOT position EQUAL 0)
      message(FATAL_ERROR "ORB dependency escaped installed prefix: ${path}")
    endif()
  endforeach()
endforeach()
add_executable(orb-contract contract.cpp)
if(DEFINED LEGACY_HISTORY_INCLUDE)
  target_include_directories(orb-contract BEFORE PRIVATE "${LEGACY_HISTORY_INCLUDE}")
endif()
target_compile_features(orb-contract PRIVATE cxx_std_17)
target_compile_options(orb-contract PRIVATE -Wall -Wextra)
target_link_libraries(orb-contract PRIVATE ORB_SLAM3::ORB_SLAM3)
if(DEFINED UPSTREAM_SHUTDOWN_SOURCE)
  add_library(upstream-shutdown SHARED "${UPSTREAM_SHUTDOWN_SOURCE}")
  target_link_libraries(upstream-shutdown PRIVATE ORB_SLAM3::ORB_SLAM3)
endif()
if(DEFINED WRONG_STOP_ORDER_SOURCE)
  add_library(wrong-stop-order SHARED "${WRONG_STOP_ORDER_SOURCE}")
  target_link_libraries(wrong-stop-order PRIVATE ORB_SLAM3::ORB_SLAM3)
endif()
foreach(case features viewer gba-join gba-stop-order atomic-stop self-join unavailable-pose shutdown)
  add_test(NAME orb-${case} COMMAND orb-contract ${case}
    "${PROBE_PREFIX}/share/orb-slam3/ORBvoc.txt"
    "${PROBE_PREFIX}/share/orb-slam3/settings/TUM1.yaml")
  set_tests_properties(orb-${case} PROPERTIES TIMEOUT 120)
endforeach()
