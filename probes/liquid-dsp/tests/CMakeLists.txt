cmake_minimum_required(VERSION 3.18)
project(ember_liquid_contract C)
enable_testing()
if(NOT IS_ABSOLUTE "${PROBE_PREFIX}")
    message(FATAL_ERROR "An absolute PROBE_PREFIX is required")
endif()
unset(liquid_DIR CACHE)
find_package(liquid 1.8.3 EXACT CONFIG REQUIRED PATHS "${PROBE_PREFIX}/lib/cmake/liquid" NO_DEFAULT_PATH)
add_executable(liquid-contract contract.c)
target_compile_features(liquid-contract PRIVATE c_std_99)
target_compile_options(liquid-contract PRIVATE -Wall -Wextra -Werror)
target_link_libraries(liquid-contract PRIVATE liquid::liquid m)
foreach(test IN ITEMS filter resample modem spectrum)
    add_test(NAME liquid-${test} COMMAND liquid-contract ${test})
    set_tests_properties(liquid-${test} PROPERTIES TIMEOUT 30)
endforeach()
